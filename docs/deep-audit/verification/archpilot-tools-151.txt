#!/usr/bin/env bash
# Background checks that raise a notification when something wants attention.
#
#   blackwall-watch --once          run every enabled check now, then exit
#   blackwall-watch --daemon        run forever (started from exec.conf)
#   blackwall-watch --list          show every check, its tier and its state
#   blackwall-watch --check NAME    run one check, ignoring its cooldown
#
# Tiers, as agreed:
#   1  safe, high signal   - always on
#   2  useful, judgement   - on, but tuned to stay quiet
#   3  clever, noisy       - on by request; expect to turn these off
#
# Set TIERS to choose, e.g. TIERS="1 2" to drop the noisy ones.
#
# Every check is cheap and read-only. Nothing here changes the system; the
# actions offered on a notification are what change things, and only when
# clicked.
set -uo pipefail

HERE="$(cd "$(dirname "$(readlink -f "$0")")" && pwd)"
STATE="${XDG_CACHE_HOME:-$HOME/.cache}/blackwall/watch"
TIERS="${TIERS:-1 2 3}"
LOOP_SECONDS="${LOOP_SECONDS:-300}"
mkdir -p "$STATE"

have() { command -v "$1" >/dev/null 2>&1; }

# A check fires at most once per cooldown, so a disk that stays full does not
# notify every five minutes forever. The state file also carries the last
# message, so a *changed* condition can re-notify before the cooldown expires.
should_fire() {  # name, cooldown_seconds, message
    local name="$1" cooldown="$2" msg="$3"
    local f="$STATE/$name"
    if [[ -f "$f" ]]; then
        local when prev
        when="$(head -1 "$f" 2>/dev/null)"
        prev="$(tail -n +2 "$f" 2>/dev/null)"
        local age=$(( $(date +%s) - ${when:-0} ))
        [[ "$prev" == "$msg" ]] && (( age < cooldown )) && return 1
    fi
    { date +%s; printf '%s' "$msg"; } > "$f"
    return 0
}

# notify-send -A implies --wait, so it blocks until the notification is
# dismissed or actioned. Each one therefore runs detached, or a single ignored
# notification would stall every later check.
fire() {  # urgency, title, body, [action_label, action_command]
    local urgency="$1" title="$2" body="$3" label="${4:-}" cmd="${5:-}"
    if [[ -n "$label" ]]; then
        (
            # The category tags this for the mako rule that stops it timing
            # out - an action button you cannot reach in six seconds is not an
            # action button.
            # Named "default" on purpose: mako binds left-click to the
            # action called `default`, so this makes clicking the notification
            # run the thing. Any other name is only reachable via
            # `makoctl invoke <name>`, which nobody is going to type.
            chosen="$(notify-send -u "$urgency" -A "default=$label" \
                      --hint=string:category:blackwall-action \
                      "$title" "$body" 2>/dev/null)"
            if [[ "$chosen" == "default" ]]; then
                # Feedback the moment it is clicked. Several of these actions
                # run for a minute with no window of their own, and a click
                # that appears to do nothing gets clicked again.
                notify-send -u low "$title" "starting: $label" >/dev/null 2>&1
                eval "$cmd"
            fi
        ) >/dev/null 2>&1 &
    else
        notify-send -u "$urgency" "$title" "$body" >/dev/null 2>&1
    fi
}

check_pacnew() {
    local files n
    files="$(find /etc -name '*.pacnew' -o -name '*.pacsave' 2>/dev/null | head -20)"
    [[ -n "$files" ]] || return 1
    n="$(printf '%s\n' "$files" | wc -l)"
    should_fire pacnew "$COOL_pacnew" "$n" || return 1
    fire normal "$n config file(s) need merging" \
        "$(printf '%s\n' "$files" | head -5 | xargs -n1 basename | paste -sd', ')" \
        "list" "kitty -e sh -c 'find /etc -name \"*.pacnew\" -o -name \"*.pacsave\"; read'"
}

    local status
    status="$(timeout 5 "$ap" waybar 2>/dev/null | head -1 \
              | python3 -c 'import json,sys
try: print(json.load(sys.stdin).get("class",""))

    should_fire archpilot "$COOL_archpilot" "waiting" || return 1

    # that ArchPilot is blocked on you.

        "open" "$ap ui --toggle"
}
