Supports animated gifs and putting different stuff in different monitors. I also did my best to make

Note `awww` will only work in a compositor that implements the layer-shell protocol. Typically,

Usage: awww <COMMAND>

Commands:
  clear        Fills the specified outputs with the given color

  clear-cache  Clears the awww cache

  unpause      Unpauses the daemon

start_binary() {                      # start_binary NAME CMD...
    local name="$1"; shift
    binary_up "$name" && { ((status_only)) && say "$name up"; return 0; }
    ((status_only)) && { say "$name DOWN"; return 0; }
    setsid "$@" >/dev/null 2>&1 9>&- &
    say "started $name"
}

start_script() {                      # start_script TAG MATCH CMD...
    local tag="$1" match="$2"; shift 2
    script_up "$tag" "$match" && { ((status_only)) && say "$tag up"; return 0; }
    ((status_only)) && { say "$tag DOWN"; return 0; }
    setsid "$@" >/dev/null 2>&1 9>&- &
    echo $! > "$RUN/$tag.pid"
    say "started $tag"
}
