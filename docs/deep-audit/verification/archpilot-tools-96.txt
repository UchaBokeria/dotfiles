94:def _wl_copy(text: str, primary: bool = False) -> None:
95:    """Hand text to wl-copy without waiting on it.
96:
97:    wl-copy forks and stays resident to serve the selection, so waiting for it
98:    to exit blocks until something else takes ownership - which froze the UI.
99:    Write, close stdin, walk away.
100:    """
101:    argv = ["wl-copy"] + (["--primary"] if primary else [])
102:    try:
103:        proc = subprocess.Popen(argv, stdin=subprocess.PIPE,
104:                                stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
105:                                start_new_session=True)
106:    except OSError:
107:        return
108:    try:
109:        proc.stdin.write(text.encode())
110:        proc.stdin.close()
111:    except (OSError, BrokenPipeError):
112:        pass
