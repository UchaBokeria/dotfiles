wifi_psk() {
    # Addressed by profile, not by SSID: the two differ whenever
    # NetworkManager has had to rename one. See wifi_profiles.
    local profile
    profile="$(wifi_profiles "$1" | head -1)"
    [[ -n "$profile" ]] || return 0
    nmcli -s -g 802-11-wireless-security.psk connection show "$profile" 2>/dev/null |
        grep -v "^Warning:" | head -1
}

# The QR encodes the standard WIFI: URI every phone camera understands, so
# sharing a network is pointing a camera at the screen rather than reading a
# password aloud.
wifi_qr() {
    local ssid="$1" psk sec tmp
    psk="$(wifi_psk "$ssid")"
    sec="WPA"; [[ -z "$psk" ]] && sec="nopass"
    tmp="$(mktemp --suffix=.png)"
    qrencode -s 8 -m 2 -o "$tmp" "WIFI:T:${sec};S:${ssid};P:${psk};;" 2>/dev/null || {
        notify-send -a blackwall "wifi" "qrencode is not installed"; return; }
    # imv/swayimg if present, otherwise hand it to xdg-open.
    for viewer in imv swayimg xdg-open; do
        command -v "$viewer" >/dev/null && { "$viewer" "$tmp" >/dev/null 2>&1 & return; }
    done
}

wifi_state() {
    have nmcli || { echo '{"on":false,"ssid":"","signal":0}'; return; }
    local on ssid signal
    on="$(nmcli -t -f WIFI general status 2>/dev/null | grep -q enabled && echo true || echo false)"
    ssid="$(nmcli -t -f NAME,TYPE connection show --active 2>/dev/null \
            | awk -F: '$2=="802-11-wireless"{print $1; exit}')"
    signal="$(nmcli -t -f IN-USE,SIGNAL device wifi 2>/dev/null \
              | awk -F: '$1=="*"{print $2; exit}')"
    printf '{"on":%s,"ssid":%s,"signal":%d}' \
        "$on" "$(printf '%s' "${ssid:-}" | jstr)" "${signal:-0}"
}

eth_state() {
    have nmcli || { echo '{"on":false,"name":""}'; return; }
    local name
    name="$(nmcli -t -f TYPE,STATE,CONNECTION device 2>/dev/null \
            | awk -F: '$1=="ethernet" && $2=="connected"{print $3; exit}')"
    printf '{"on":%s,"name":%s}' \
        "$([[ -n "$name" ]] && echo true || echo false)" \
        "$(printf '%s' "${name:-}" | jstr)"
}

vpn_state() {
    have tailscale || { echo '{"on":false,"name":"","ip":""}'; return; }
    timeout 5 tailscale status --json 2>/dev/null | python3 -c '
import json, sys
try:
    d = json.load(sys.stdin)
except Exception:
    print(json.dumps({"on": False, "name": "", "ip": ""})); raise SystemExit
me = d.get("Self") or {}
print(json.dumps({
    "on": d.get("BackendState") == "Running",
    "name": me.get("HostName", ""),
    "ip": (me.get("TailscaleIPs") or [""])[0],
    "peers": len(d.get("Peer") or {}),
}))' 2>/dev/null || echo '{"on":false,"name":"","ip":""}'
}

bt_state() {
    have bluetoothctl || { echo '{"on":false,"connected":""}'; return; }
    local powered connected
    powered="$(timeout 5 bluetoothctl show 2>/dev/null | grep -oP 'Powered: \K\w+')"
    connected="$(timeout 5 bluetoothctl devices Connected 2>/dev/null \
                 | head -1 | cut -d' ' -f3-)"
    printf '{"on":%s,"connected":%s}' \
        "$([[ "$powered" == yes ]] && echo true || echo false)" \
        "$(printf '%s' "${connected:-}" | jstr)"
}
