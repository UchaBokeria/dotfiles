;; Notification history.
;;
;; Backed by scripts/blackwall-notifyd - a passive D-Bus monitor - rather than
;; by the notification daemon itself. That is deliberate: mako's buffer held a
;; handful of entries with no timestamps and emptied on every restart, and
;; swaync's own history lives inside swaync. Recording independently means the
;; history survives swapping the daemon underneath it, which is exactly what
;; just happened.
;;
;; Popups, action buttons and do-not-disturb are swaync's job. This is the
;; record, and what you can do to an application because of it.

(defvar notif_open "")
;; "list" | "settings"
(defvar notif_view "list")

(defpoll notif_history
  :initial "[]"
  :interval "10s"
  :run-while is_control_center_open
  `~/.config/.dotfiles/blackwall/scripts/blackwall-notify history 40`)

(defpoll notif_dnd
  :initial "false"
  :interval "2s"
  :run-while is_control_center_open
  ;; Asks whichever daemon is live. Polling swaync-client directly meant the
  ;; icon could never change while mako was still the daemon.
  `~/.config/.dotfiles/blackwall/scripts/blackwall-quiet dnd-state`)

(defpoll notif_apps
  :initial "[]"
  :interval "8s"
  :run-while is_control_center_open
  `~/.config/.dotfiles/blackwall/scripts/blackwall-quiet apps`)

(defpoll notif_silenced
  :initial "[]"
  :interval "5s"
  :run-while is_control_center_open
  `~/.config/.dotfiles/blackwall/scripts/blackwall-quiet silenced`)

(defwidget silenced_row [entry]
  (box :class "silenced-row" :orientation "h" :space-evenly false :spacing 8
    (box :orientation "v" :space-evenly false :hexpand true
      (label :class "list-name" :text {entry.app} :xalign 0 :limit-width 24)
      (label :class "list-meta" :xalign 0
             :text {entry.state == "ignored" ? "ignored · ${entry.left}"
                                             : "muted · ${entry.left}"}))
    (eventbox :cursor "hand2"
      (button :class "notif-act"
              :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-quiet unmute '${entry.app}'"
        "unsilence"))))

(defwidget app_silence_row [entry]
  ;; The application's own glyph, the same one its notifications carry, and
  ;; the three durations as one segmented control rather than three separate
  ;; pills with gaps between them.
  (box :class "silenced-row" :orientation "h" :space-evenly false :spacing 9
    (box :class "app-badge ${entry.silenced ? "on" : ""}" :valign "center"
      (label :class "app-badge-icon" :text {entry.glyph}))
    (label :class "list-name" :text {entry.app} :xalign 0 :hexpand true :limit-width 18)
    (box :class "seg" :orientation "h" :space-evenly false
      (eventbox :cursor "hand2"
        (button :class "seg-btn ${entry.silenced ? "on" : ""}"
                :onclick {entry.silenced
                          ? "~/.config/.dotfiles/blackwall/scripts/blackwall-quiet unmute '${entry.app}'"
                          : "~/.config/.dotfiles/blackwall/scripts/blackwall-quiet mute '${entry.app}' 1h"}
          {entry.silenced ? "on" : "1h"}))
      (eventbox :cursor "hand2"
        (button :class "seg-btn"
                :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-quiet mute '${entry.app}' 1d"
          "1d"))
      (eventbox :cursor "hand2"
        (button :class "seg-btn danger"
                :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-quiet ignore '${entry.app}' forever"
          "never")))))

(defwidget notif_settings []
  ;; Everything fits without scrolling now. It did not before: a heading and an
  ;; empty-state line for "silenced" were drawn even with nothing silenced,
  ;; which is the usual case, and that section plus a two-line footnote pushed
  ;; the application list past the bottom.
  (box :orientation "v" :space-evenly false :spacing 8

    (box :class "net-back-row" :orientation "h" :space-evenly false :spacing 8
      (eventbox :cursor "hand2"
        (button :class "net-back" :onclick "eww update notif_view=list" "‹"))
      (label :class "net-back-title" :text "Notification settings"
             :xalign 0 :hexpand true))

    (box :class "vpn-settings" :orientation "v" :space-evenly false :spacing 2
      (box :class "vpn-opt" :orientation "h" :space-evenly false :spacing 9
        (box :class "app-badge ${notif_dnd == "true" ? "on" : ""}" :valign "center"
          (label :class "app-badge-icon" :text {notif_dnd == "true" ? "󰂛" : "󰂚"}))
        (box :orientation "v" :space-evenly false :hexpand true
          (label :class "list-name" :text "Do not disturb" :xalign 0)
          (label :class "list-meta" :xalign 0
                 :text {notif_dnd == "true" ? "nothing appears on screen"
                                            : "notifications appear normally"}))
        (eventbox :cursor "hand2"
          (button :class "vpn-switch ${notif_dnd == "true" ? "on" : ""}"
                  :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-quiet dnd toggle"
            {notif_dnd == "true" ? "on" : "off"}))))

    ;; Only when something is silenced. A heading over "no application is
    ;; silenced" spent two lines saying nothing on every visit.
    (for _ in {arraylength(notif_silenced) > 0 ? [1] : []}
      (box :orientation "v" :space-evenly false :spacing 6
        (label :class "section-title" :xalign 0 :text "Silenced" :hexpand true)
        (box :class "vpn-settings" :orientation "v" :space-evenly false :spacing 2
          (for entry in notif_silenced (silenced_row :entry entry)))))

    (label :class "section-title" :xalign 0 :text "Applications" :hexpand true)

    ;; Buttons in the panel rather than a rofi picker. A picker meant two
    ;; prompts launched from inside a click handler, and each of those is a
    ;; place for the flow to end without saying anything.
    (box :class "vpn-settings" :orientation "v" :space-evenly false :spacing 2
      (for a in notif_apps (app_silence_row :entry a)))

    (label :class "cal-empty-hint" :xalign 0 :wrap true :show-truncated false
           :text "muted hides it and keeps it here; never drops it entirely")))

(defwidget notif_entry [entry index]
  (box :class "notif ${notif_open == "${index}" ? "open" : ""}"
       :orientation "v" :space-evenly false

    (eventbox :cursor "hand2"
      (button :class "notif-body-btn"
              :onclick "eww update notif_open=${notif_open == "${index}" ? "-" : index}"
        ;; A badge, then the text. The badge replaced a 7px dot that carried
        ;; urgency and nothing else: it now says which application this came
        ;; from as well, so the app name no longer has to be the first thing
        ;; you read on every row. Urgency stays, as the badge's colour.
        (box :orientation "h" :space-evenly false :spacing 11
          (box :class "notif-badge ${entry.urgency}" :valign "start"
            (label :class "notif-badge-icon" :text {entry.glyph}))

          (box :orientation "v" :space-evenly false :spacing 2 :hexpand true

            (box :class "notif-meta" :orientation "h" :space-evenly false :spacing 7
              (label :class "notif-app" :text {entry.app} :xalign 0 :limit-width 20)
              (label :class "notif-ago" :text {entry.ago} :hexpand true :xalign 1))

            (label :class "notif-summary" :text {entry.summary} :xalign 0 :hexpand true
                   :limit-width 34 :visible {entry.summary != ""})

            (label :class "notif-body" :text {entry.body} :xalign 0 :hexpand true
                   :limit-width {notif_open == "${index}" ? 200 : 40}
                   :wrap {notif_open == "${index}"}
                   :show-truncated {notif_open != "${index}"}
                   :visible {entry.body != ""})))))

    (revealer :transition "slidedown" :duration "180ms"
              :reveal {notif_open == "${index}"}
      (box :class "notif-actions" :orientation "h" :space-evenly false :spacing 6
        (eventbox :cursor "hand2"
          (button :class "notif-act"
                  :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-notify drop ${index} && eww update notif_open='' notif_history=\"$(~/.config/.dotfiles/blackwall/scripts/blackwall-notify history 40)\""
            "dismiss"))
        (eventbox :cursor "hand2"
          (button :class "notif-act"
                  :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-quiet mute '${entry.app}' 1h"
            "mute 1h"))
        (eventbox :cursor "hand2"
          (button :class "notif-act"
                  :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-quiet mute '${entry.app}' 1d"
            "1d"))
        (eventbox :cursor "hand2"
          (button :class "notif-act danger"
                  :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-quiet ignore '${entry.app}' forever"
            "never"))))))
