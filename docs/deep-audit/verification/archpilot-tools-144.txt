#!/bin/bash
# Put the wallpaper back after a login.
#
# The old one line here was:
#
#   LAST_WALL="$(cat ~/.cache/last_wallpaper > /dev/null 2>&1 || echo DEFAULT)"
#
# The redirect applies to `cat`, not to the substitution, so whenever the cache
# file existed the path went to /dev/null and LAST_WALL came out EMPTY - and
# `awww img ""` fails. That is why the desktop came up with Hyprland's own
# default picture until something re-ran setwall.
#
# awww-daemon also loses a race at login: it is started here and needs a moment
# before it will accept an image, and it has crashed outright at boot more than
# once (two SIGABRTs on 2026-09-12). So this retries instead of firing once.
set -u

CACHE="$HOME/.cache/last_wallpaper"
DEFAULT="$HOME/.config/wallpapers/cyberpunk.png"

;; Volume, microphone and brightness.
;;
;; Each audio row is icon / level / slider / device, where the icon is a mute
;; toggle and the device name opens a picker. The old version had a decorative
;; glyph and a slider, so the only thing you could do here was set a level -
;; muting or changing output device meant leaving the panel.

(include "state/ios.yuck")

(defwidget audio_row [icon level silent desc kind ?loopback]
  (box :class "vol ${silent ? "silent" : ""}"
       :orientation "h"
       :space-evenly false
       :spacing 10

    ;; The icon is the mute button. It is the control people reach for first
    ;; and it was previously the one thing on the row that did nothing.
    (eventbox :cursor "hand2"
      (button :class "vol-icon ${silent ? "silent" : ""}"
              :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-audio mute-${kind}"
        icon))

    (label :class "vol-level" :text "${level}%" :xalign 0 :width 34)

    (eventbox :cursor "hand2" :hexpand true
      (scale :class "vol-scale"
             :min 0 :max 100
             :value level
             :draw-value false
             :hexpand true
             :onchange "wpctl set-volume @DEFAULT_AUDIO_${kind == "sink" ? "SINK" : "SOURCE"}@ {}%"))

    ;; Device name doubles as the disclosure control - a separate chevron would
    ;; be a second target for the same action.
    (eventbox :cursor "hand2"
      (button :class "vol-device ${audio_expanded == kind ? "on" : ""}"
              :onclick "eww update audio_expanded=${audio_expanded == kind ? "-" : kind}"
              :tooltip {loopback ?: false
                        ? "${desc} - a loopback of the output, not a microphone"
                        : desc}
        (label :class "vol-device-label" :text desc :limit-width 10 :xalign 1)))))

(defwidget audio_device [entry kind]
  (eventbox :cursor "hand2"
    (button :class "list-row ${entry.active ? "on" : ""}"
            :onclick "~/.config/.dotfiles/blackwall/scripts/blackwall-audio set-${kind} '${entry.name}'"
      (box :orientation "h" :space-evenly false :spacing 10
        (label :class "list-mark" :text {entry.active ? "●" : "○"})
        (label :class "list-name" :text {entry.desc} :hexpand true
               :halign "start" :limit-width 24)
        (label :class "list-meta" :text {entry.loopback ? "loopback" : ""})))))
