;; System load, as four small gauges.
;;
;; These were `circular-progress` widgets with no size and no stylesheet, so
;; they expanded to fill a 300px box and rendered as enormous broken arcs. A
;; gauge only works when it is small enough to read at a glance beside its
;; number - at 300px it stops being a gauge and becomes abstract art.
;;
;; No glyphs on the gauges - deliberately, not because they cannot render.
;; They could not, for a while: `* { font-family: $font_ui }` in eww.scss sets
;; the family directly on every label, which beats inheriting it from a styled
;; parent, so an icon class on a container never reached the text. That is
;; fixed (see .vol-icon in eww.scss, and $font_icon).
;;
;; They are still absent because a gauge with the word "cpu" under it does not
;; also need a picture of a cpu. The label is the clearer of the two.

(defwidget stat_gauge [label value ?class]
  (box :orientation "v"
       :space-evenly false
       :spacing 6
       :class "stat"
       :halign "center"

    ;; circular-progress reports no natural size, so it has to be given one by
    ;; a box around it - CSS min-width on the widget itself is ignored.
    (box :class "gauge-slot" :width 54 :height 54 :halign "center"
      (circular-progress
        :class "gauge ${class ?: ""}"
        :value {value > 100 ? 100 : (value < 0 ? 0 : value)}
        :thickness 5
        :start-at 75))

    (label :class "gauge-value" :text "${round(value, 0)}%")
    (label :class "stat-label" :text label)))

(defwidget control_center_sys [?cpu ?ram ?storage ?net]
  (box :orientation "v"
       :space-evenly false
       :spacing 10
       :class "bar-widget-section"

    (label :class "section-title" :xalign 0 :text "System" :hexpand true)

    ;; Three rings, then the network on its own line.
    ;;
    ;; It used to be a fourth cell in this row: a bare number in a 54px gauge
    ;; slot with "KB/s" under it. That was wrong twice over. A rate has no
    ;; percentage, so it had a ring's worth of space and nothing to draw in it;
    ;; and the number was rx only while the label just said "network", so an
    ;; upload showed as zero. A rate wants a shape over time, and that does not
    ;; fit in a square meant for a dial.
    (box :orientation "h" :space-evenly true :spacing 8
      (stat_gauge :label "cpu"  :value cpu)
      (stat_gauge :label "ram"  :value ram)
      (stat_gauge :label "disk" :value storage
                  :class {storage > 85 ? "warn" : ""}))

    (box :class "net-strip" :orientation "v" :space-evenly false :spacing 6
      (box :orientation "h" :space-evenly false :spacing 8
        (label :class "stat-label" :text {net.iface != "" ? net.iface : "network"}
               :xalign 0 :hexpand true)
        (label :class "net-down" :text "↓ ${net.down}")
        (label :class "net-up"   :text "↑ ${net.up}")
        (label :class "net-unit" :text "KB/s"))

      ;; eww's own graph, rather than the block characters this used to be.
      ;; Those were sized for a gauge slot, and drew their colours by grepping
      ;; colors.scss on a relative path - which resolved against whatever
      ;; directory eww happened to be in, so the spans came out with an empty
      ;; foreground and the bars were whatever colour the label already was.
      (graph :class "net-graph"
             :value {net.down + net.up}
             :time-range "90s"
             :thickness 2
             :line-style "round"
             :dynamic true
             :height 44
             :hexpand true))))

/* A raised pane within the panel: one step forward, its own hairline. */
.bar-widget-section {
  /* Glass with a lit top edge, not a flat fill inside a hairline ring.
     
     The ring was a second border a few pixels inside the panel's own, and two
     concentric outlines around the same content read as a box someone forgot
     to remove. Depth does the separating instead: a soft vertical gradient and
     a single highlight along the top, so the section sits *in* the panel
     rather than being drawn on it.
     
     This is a container, which is why it gets a gradient while the controls
     stay flat - the quick action tiles were flattened for the opposite reason,
     that a control lit like a surface reads as a different design. */
  background: linear-gradient(160deg, $glass_hi 0%, $glass 55%, $glass_lo 100%);
  box-shadow: inset 0 1px 0 $rim_top;
  border: none;
  border-radius: $r_md;
  padding: $s_lg;
  margin-top: $s_lg;
}
