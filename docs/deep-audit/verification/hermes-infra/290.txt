#!/usr/bin/env bash
# session-mgr.sh — multi-session lifecycle for the cloud Claude.
#
# MODEL
#   The session that owns the Telegram poller is ALWAYS the tmux session named
#   "claude" ("attached"). Telegram allows exactly one getUpdates consumer per
#   bot token, so only one session can be attached at a time.
#   Detached sessions keep running under tmux names `cl-<key>-<HHMM>`; they have
#   no poller but are still bridged to claude.ai (remoteControlAtStartup), so
#   Ucha continues them from the MOBILE APP. That is the whole point: Telegram
#   spawns/switches sessions, the app is where long work continues.
#
# Verified behaviour this relies on (2026-08-20):
#   - killing a session's `bun server.ts` poller leaves claude + tmux alive and
#     it does NOT respawn  → clean detach
#   - a newly started session with --channels claims the poller while the old
#     session keeps running → clean attach
#
# COMMANDS
#   detach              rename the attached session to cl-<key>-<HHMM>, kill its
#                       poller; it stays alive for the app
#   new [key]           detach current, start a FRESH attached session (same
#                       project unless a key is given)
#   attach <target>     detach current, then make <target> the attached session.
#                       <target> = tmux name of a live detached session, or a
#                       transcript/session UUID to resume.
#   retire              shut down the oldest detached session (RAM cap)
#   list                machine-readable list of live sessions

#!/usr/bin/env bash
# Watchdog for the cloud Claude session + Telegram channel poller.
#
# Why this exists: on 2026-08-19 the bot was found dead since 2026-07-18 — the
# plugin's `bun server.ts` (spawned by claude's --channels flag) had exited and
# nothing noticed. claude-session.sh runs only @reboot and the host had 47 days
# uptime, so the session sat alive-but-deaf for a month. The same audit found
# the OAuth login had ALSO expired, which no restart can repair.
#
# Cron: */5 * * * *  — see docs/telegram-widget.md.
#
# Three distinct faults, three different responses:
#   session gone   -> restart immediately (no context left to lose)
#   poller gone    -> restart after N consecutive misses (restart WIPES context,
#                     so we pay that price only for a persistent fault)
#   login expired  -> alert only; restarting cannot fix it, /login is interactive
set -uo pipefail
export PATH="/home/scriptkid/.nvm/versions/node/v22.23.1/bin:/home/scriptkid/.bun/bin:/home/scriptkid/.local/bin:/usr/bin:/bin"

CLOUD=/home/scriptkid/opt/cloud
STATE=$CLOUD/tmp/poller-watchdog
LOG=$CLOUD/telegram/watchdog.log
SESSION_SH=$CLOUD/telegram/claude-session.sh
FAIL_AFTER=3          # consecutive misses before a context-destroying restart
RESTART_COOLDOWN=1800 # 30 min minimum between restarts
ALERT_COOLDOWN=21600  # 6 h between repeat alerts of the same kind
mkdir -p "$STATE"

log() { echo "$(date -Is) $*" >>"$LOG"; }
now() { date +%s; }
