# Runs `docker compose <action>` for a dev project and posts the result to

send() {  # <text> — Markdown, falling back to plain if Telegram rejects the markup
  [ -n "$TOKEN" ] || return
  local resp mid
  resp=$(curl -s "https://api.telegram.org/bot$TOKEN/sendMessage" \
    --data-urlencode "chat_id=$CHAT" --data-urlencode "text=$1" -d parse_mode=Markdown)
  mid=$(printf '%s' "$resp" | grep -o '"message_id":[0-9]*' | head -1 | cut -d: -f2)
  if [ -z "$mid" ]; then
    resp=$(curl -s "https://api.telegram.org/bot$TOKEN/sendMessage" \
      --data-urlencode "chat_id=$CHAT" --data-urlencode "text=$1")
    mid=$(printf '%s' "$resp" | grep -o '"message_id":[0-9]*' | head -1 | cut -d: -f2)
  fi
  [ -n "$mid" ] && printf '%s\t%s\n' "$CHAT" "$mid" >> "$LOG"
}

case "$ACTION" in
  up)      CMD=(up -d --remove-orphans);;
  down)    CMD=(down);;
  restart) CMD=(restart);;
  status)  CMD=(ps);;
  *) send "⚠️ svc: unknown action '$ACTION'"; exit 1;;
esac
[ -d "$DIR" ] || { send "⚠️ *$LABEL*: path not found: $DIR"; exit 1; }
cd "$DIR" || { send "⚠️ *$LABEL*: cannot cd into $DIR"; exit 1; }

send "⏳ *$LABEL* — docker compose $ACTION…"
OUT=$(docker compose "${CMD[@]}" 2>&1); RC=$?
TAIL=$(printf '%s' "$OUT" | tail -c 2500)
[ -z "$TAIL" ] && TAIL="(no output)"
ICON=$([ $RC -eq 0 ] && echo "✅" || echo "❌")
send "$ICON *$LABEL* — docker compose $ACTION (rc=$RC)
\`\`\`
$TAIL
\`\`\`"
