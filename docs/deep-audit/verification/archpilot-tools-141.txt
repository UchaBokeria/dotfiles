# moment playback stops - so `apps` goes empty and the panel looked broken

media_state() {
    have playerctl || { echo '{"has":false}'; return; }
    local player=""
    while read -r p; do
        case "$p" in
            plasma-browser-integration*|kdeconnect*|playerctld*) continue ;;
        esac
        if [[ "$(playerctl -p "$p" status 2>/dev/null)" == Playing ]]; then
            player="$p"; break
        fi
        [[ -z "$player" ]] && player="$p"
    done < <(playerctl -l 2>/dev/null)
    [[ -z "$player" ]] && { echo '{"has":false}'; return; }

    python3 - "$player" <<'PY_END'

                        capture_output=True, text=True).stdout.strip() or "Stopped"

    "has": bool(title or artist),
    "player": p.split(".")[0],
    "status": status.lower(),

    "title": title[:44] or "nothing",
    "artist": artist[:34],
}))
PY_END
