#!/usr/bin/env bash
# blackwall rofi launcher.
#
#   launch grid            full-screen app grid   (SUPER+R)
#   launch list [args...]  centred list           (SUPER+F, window switcher)
#   launch compact [args]  small picker           (SUPER+U, dmenu pipes)
#
# Any extra arguments are passed straight through to rofi, so this also works
# as a dmenu front end:  ... | launch compact -dmenu -p "pick"
set -euo pipefail

layout="${1:-list}"
shift || true

theme="${HOME}/.config/rofi/blackwall/${layout}.rasi"
[[ -f "$theme" ]] || { echo "no such layout: $layout" >&2; exit 1; }

# With no arguments, decide between launcher and dmenu by asking whether
# anything is actually being piped in - a pipe or a redirected file - rather
# than by asking whether stdin is a terminal. Under a Hyprland `exec` stdin
# happens to be /dev/tty1, so the terminal test gives the right answer there by
# luck; run the same script from a service, a cron job or another script and
# stdin is /dev/null, which is not a terminal and not a pipe either, and the
# terminal test would silently open an empty dmenu instead of the launcher.
if (($# == 0)); then
    if [[ -p /dev/stdin || -f /dev/stdin ]]; then
        set -- -dmenu
    else
        set -- -show drun
    fi
